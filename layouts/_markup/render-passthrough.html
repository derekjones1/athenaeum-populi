{{- /* Athenaeum Populi — build-time KaTeX passthrough render hook. Math is rendered by Hugo's pinned transform.ToMath implementation and ships as static HTML+MathML. There is intentionally no client-side/CDN fallback: an authoring error must fail the build instead of becoming a runtime fetch. */ -}} {{- $engine := site.Params.math.engine | default "katex" -}} {{- if eq $engine "katex" -}} {{- $opts := dict "output" "htmlAndMathml" "displayMode" (eq .Type "block") -}} {{- with try (transform.ToMath .Inner $opts) -}} {{- with .Err -}} {{- errorf "transform.ToMath failed to render math (%s): %s" $.Position . -}} {{- else -}} {{- .Value -}} {{- $.Page.Store.Set "hasMath" true -}} {{- end -}} {{- end -}} {{- else -}} {{- errorf "unsupported math engine %q (%s)" $engine .Position -}} {{- end -}}
