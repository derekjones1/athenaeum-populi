{{- /* Athenaeum Populi — custom head-end hook (Hextra calls this at ). Builds the interactive Web Components bundle from assets/js/main.js with Hugo Pipes. Math is already rendered to static HTML+MathML at build time. */ -}} {{- /* Interactive Web Components (esbuild via Hugo Pipes js.Build). Two bundles: - fillin-engine.js : MathLive + Compute Engine grader (~1.9 MB). Built on its own; loaded at RUNTIME only on pages with a . - main.js (components entry): tiny; registers the custom elements. The engine's URL is injected via `params` (read as @params) so the entry never bundles MathLive. js.Build has no working `splitting` option, hence the two-bundle split. */ -}} {{- if site.Params.components.enable -}} {{- $engineURL := "" -}} {{- with resources.Get "js/fillin-engine.js" -}} {{- $eopts := dict "minify" hugo.IsProduction "targetPath" "js/fillin-engine.js" "format" "esm" -}} {{- with try (. | js.Build $eopts) -}} {{- with .Err -}}{{- errorf "js.Build failed for fillin-engine: %s" . -}} {{- else -}}{{- $engineURL = .Value.RelPermalink -}}{{- end -}} {{- end -}} {{- end -}} {{- $graphplotURL := "" -}} {{- with resources.Get "js/graphplot-engine.js" -}} {{- $gopts := dict "minify" hugo.IsProduction "targetPath" "js/graphplot-engine.js" "format" "esm" -}} {{- with try (. | js.Build $gopts) -}} {{- with .Err -}}{{- errorf "js.Build failed for graphplot-engine: %s" . -}} {{- else -}}{{- $graphplotURL = .Value.RelPermalink -}}{{- end -}} {{- end -}} {{- end -}} {{- with resources.Get "js/main.js" -}} {{- $opts := dict "minify" hugo.IsProduction "targetPath" "js/components.js" "format" "esm" "params" (dict "engineUrl" $engineURL "graphplotEngineUrl" $graphplotURL) -}} {{- with try (. | js.Build $opts) -}} {{- with .Err -}} {{- errorf "js.Build failed for components bundle: %s" . -}} {{- else -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}
