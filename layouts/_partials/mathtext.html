{{- /* mathtext — render a shortcode string that mixes prose and inline $...$ math. Interactive exercise shortcodes use it for question, hint, and answerDisplay text so math is typeset at build time with no flash of raw LaTeX. Rules: - `$...$` runs are rendered with KaTeX (inline mode). - `\$` is a literal dollar sign (money), never a delimiter. - prose runs are rendered as inline Markdown (goldmark, unsafe HTML on, exactly like body prose), so `*Drosophila*` italicizes and an allele superscript `IA` renders as a superscript instead of the literal tag text the old htmlEscape produced. Aria strings built from these params must go through plainify (see multiplechoice/selfcheck). Call: {{ partial "mathtext" "Simplify $\tfrac{3}{4}$ for \$5" }} Returns trusted HTML. */ -}} {{- $src := . -}} {{- /* Sentinel must survive goldmark: a NUL byte is sanitized to U+FFFD by the markdown renderer, so use a private-use codepoint instead. */ -}} {{- $NUL := "\uE000" -}} {{- /* protect escaped dollars so they can't pair as delimiters */ -}} {{- $protected := replace $src "\\$" $NUL -}} {{- $parts := split $protected "$" -}} {{- $sb := slice -}} {{- range $i, $p := $parts -}} {{- if eq (mod $i 2) 1 -}} {{- $tex := replace $p $NUL "\\$" -}} {{- $opts := dict "throwOnError" false "output" "htmlAndMathml" "displayMode" false -}} {{- with try (transform.ToMath $tex $opts) -}} {{- with .Err -}} {{- errorf "mathtext: KaTeX failed on %q: %s" $tex . -}} {{- else -}} {{- $sb = $sb | append (printf "%s" .Value) -}} {{- end -}} {{- end -}} {{- else -}} {{- /* Markdownify with the escaped dollars still behind the sentinel: a bare `$` restored first would re-enter goldmark's math passthrough and KaTeX-error on money amounts. Restore after. */ -}} {{- /* markdownify trims its input, which ate the space on each side of every math run ("Evaluate $x$ when" → "Evaluatexwhen"). Re-attach the run's own edge whitespace; a whitespace-only run passes as-is. */ -}} {{- if eq (trim $p " \t\n\r") "" -}} {{- $sb = $sb | append $p -}} {{- else -}} {{- $lead := replaceRE `^(\s*)[\s\S]*$` "$1" $p -}} {{- $trail := replaceRE `^[\s\S]*?(\s*)$` "$1" $p -}} {{- $rendered := printf "%s" ($p | markdownify) -}} {{- $sb = $sb | append (printf "%s%s%s" $lead (replace $rendered $NUL "$") $trail) -}} {{- end -}} {{- end -}} {{- end -}} {{- return (delimit $sb "" | safeHTML) -}}
