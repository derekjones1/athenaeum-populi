{{- /* mathtext — render a shortcode string that mixes prose and inline $...$ math. Interactive exercise shortcodes use it for question, hint, and answerDisplay text so math is typeset at build time with no flash of raw LaTeX. Rules: - `$...$` runs are rendered with KaTeX (inline mode). - `\$` is a literal dollar sign (money), never a delimiter. - prose runs are rendered as inline Markdown (goldmark, unsafe HTML on, exactly like body prose), so `*Drosophila*` italicizes and an allele superscript `IA` renders as a superscript instead of the literal tag text the old htmlEscape produced. Aria strings built from these params must go through plainify (see multiplechoice/selfcheck). Call: {{ partial "mathtext" "Simplify $\tfrac{3}{4}$ for \$5" }} Returns trusted HTML. */ -}} {{- $src := . -}} {{- /* Sentinel must survive goldmark: a NUL byte is sanitized to U+FFFD by the markdown renderer, so use a private-use codepoint instead. */ -}} {{- $NUL := "\uE000" -}} {{- /* protect escaped dollars so they can't pair as delimiters */ -}} {{- $protected := replace $src "\\$" $NUL -}} {{- $parts := split $protected "$" -}} {{- $sb := slice -}} {{- /* Punctuation typed straight after a math run is set inside the run's own `.ap-math-glue` span, so it cannot start the next line alone (layouts/_markup/render-passthrough.html says why, and what the between a formula's pieces is for). `$glued` is that punctuation, taken off the front of the prose part that follows. */ -}} {{- $glued := "" -}} {{- /* A label typed straight before a math run, "(ii) ", and a value typed straight after it, " = 180.5 kJ", join the same span (`_partials/math-glue-patterns.html` says which and why). `$label` is the label, taken off the end of the prose part before the run. */ -}} {{- $re := partialCached "math-glue-patterns.html" "math-glue" -}} {{- $label := "" -}} {{- range $i, $p := $parts -}} {{- if eq (mod $i 2) 1 -}} {{- $tex := replace $p $NUL "\\$" -}} {{- $opts := dict "throwOnError" false "output" "htmlAndMathml" "displayMode" false -}} {{- with try (transform.ToMath $tex $opts) -}} {{- with .Err -}} {{- errorf "mathtext: KaTeX failed on %q: %s" $tex . -}} {{- else -}} {{- /* A short number in scientific notation is held on one line, as the passthrough render hook holds it in body prose (layouts/_markup/render-passthrough.html says why): a stem that ended a line at "1.497 ×" was where the break was first seen. */ -}} {{- $math := replace (printf "%s" .Value) `` `` -}} {{- if and (le (len $tex) 60) (findRE `\\times\s*10\^` $tex 1) -}} {{- $math = printf `%s` $math -}} {{- end -}} {{- if lt (add $i 1) (len $parts) -}} {{- /* A straight quote here is an apostrophe or a closing quote ("$b$'s"); markdownify would have curled it, so it is curled by hand. */ -}} {{- with findRE (printf `^(?:%s)?[.,;:?!)’”']*` $re.value) (index $parts (add $i 1)) 1 -}} {{- $glued = index . 0 -}} {{- end -}} {{- end -}} {{- if or $glued $label -}} {{- $lead := cond (ne $label "") (printf `%s ` $label) "" -}} {{- $math = printf `%s%s%s` $lead $math (replace $glued "'" "’" | htmlEscape) -}} {{- $label = "" -}} {{- end -}} {{- $sb = $sb | append $math -}} {{- end -}} {{- end -}} {{- else -}} {{- /* Markdownify with the escaped dollars still behind the sentinel: a bare `$` restored first would re-enter goldmark's math passthrough and KaTeX-error on money amounts. Restore after. */ -}} {{- /* markdownify trims its input, which ate the space on each side of every math run ("Evaluate $x$ when" → "Evaluatexwhen"). Re-attach the run's own edge whitespace; a whitespace-only run passes as-is. */ -}} {{- if $glued -}} {{- $p = strings.TrimPrefix $glued $p -}} {{- $glued = "" -}} {{- end -}} {{- if lt (add $i 1) (len $parts) -}} {{- with findRE (printf `(?:^|\s)%s $` $re.label) $p 1 -}} {{- $label = trim (index . 0) " \t\n\r" -}} {{- $p = strings.TrimSuffix (printf "%s " $label) $p -}} {{- end -}} {{- end -}} {{- if eq (trim $p " \t\n\r") "" -}} {{- $sb = $sb | append $p -}} {{- else -}} {{- $lead := replaceRE `^(\s*)[\s\S]*$` "$1" $p -}} {{- $trail := replaceRE `^[\s\S]*?(\s*)$` "$1" $p -}} {{- /* A prose piece that follows a math run is the middle of a sentence, never the start of a block: "$P$-$V$ graph" leaves the piece "-", and "$x$ - $y$" the piece " - ", which goldmark read as an empty bullet-list item (Chemistry 9.2's "P-V graph" stems and Precalculus 8.6's "x-y" stems rendered "the P", a bullet, "V graph" — found by the rendered-page reading, November 17, 2026). A leading list, heading, or quote marker there is escaped so it stays text. */ -}} {{- $md := $p -}} {{- if gt $i 0 -}} {{- $md = replaceRE `^(\s*)([-+*>#])(\s|$)` `${1}\${2}${3}` $md -}} {{- $md = replaceRE `^(\s*\d+)([.)])(\s|$)` `${1}\${2}${3}` $md -}} {{- end -}} {{- $rendered := printf "%s" ($md | markdownify) -}} {{- $sb = $sb | append (printf "%s%s%s" $lead (replace $rendered $NUL "$") $trail) -}} {{- end -}} {{- end -}} {{- end -}} {{- return (delimit $sb "" | safeHTML) -}}
